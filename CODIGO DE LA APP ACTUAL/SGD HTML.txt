function validarSesionGuardada(){
  var t=sessionStorage.getItem('sgd_token');
  if(!t) return;
  google.script.run
    .withSuccessHandler(function(r){
      if(r&&r.success&&r.autenticado){
        currentUser=r.usuario; sessionToken=t;
        mostrarApp(); iniciarApp(); iniciarPolling();
      } else { sessionStorage.removeItem('sgd_token'); }
    })
    .withFailureHandler(function(){ sessionStorage.removeItem('sgd_token'); })
    .validarSesion(t);
}

function iniciarSesion(e){
  e.preventDefault();
  if(loginBusy) return;
  var email=document.getElementById('loginEmail').value.trim();
  var pass=document.getElementById('loginPassword').value;
  var btn=document.getElementById('loginBtn');
  var err=document.getElementById('loginError');
  loginBusy=true;
  btn.innerHTML='<i class="fas fa-spinner fa-spin"></i> VERIFICANDO...';
  btn.disabled=true;
  google.script.run
    .withSuccessHandler(function(r){
      btn.innerHTML='<i class="fas fa-sign-in-alt"></i> ACCEDER AL SISTEMA';
      btn.disabled=false; loginBusy=false;
      if(r&&r.autenticado){
        currentUser=r.usuario; sessionToken=r.token;
        sessionStorage.setItem('sgd_token',r.token);
        mostrarApp(); iniciarApp();
        mostrarModal('🎯','BIENVENIDO',currentUser.nombre,'Rol: '+currentUser.rol+'\n🔐 Sesión segura activa');
        iniciarPolling();
      } else {
        err.innerHTML='<i class="fas fa-exclamation-triangle"></i> '+(r?r.message||'Credenciales incorrectas':'Error de conexión');
        err.classList.remove('hidden');
        setTimeout(function(){ err.classList.add('hidden'); },4000);
      }
    })
    .withFailureHandler(function(){
      btn.innerHTML='<i class="fas fa-sign-in-alt"></i> ACCEDER AL SISTEMA';
      btn.disabled=false; loginBusy=false;
      err.innerHTML='<i class="fas fa-plug"></i> Error de conexión con el servidor';
      err.classList.remove('hidden');
      setTimeout(function(){ err.classList.add('hidden'); },4000);
    })
    .verificarCredenciales(email,pass);
}
function mostrarApp(){
  document.getElementById('loginScreen').style.display='none';
  document.getElementById('mainApp').classList.remove('hidden');
  document.getElementById('chatFab').classList.remove('hidden');
}
function cerrarSesion(){
  Swal.fire({title:'¿Cerrar sesión?',icon:'question',showCancelButton:true,confirmButtonText:'SÍ',cancelButtonText:'NO',background:'#ffffff',color:'#142a44',confirmButtonColor:'#ef4444'})
    .then(function(r){
      if(r.isConfirmed){
        if(pollingInterval){ clearInterval(pollingInterval); pollingInterval=null; }
        if(sessionToken){ google.script.run.cerrarSesionToken(sessionToken); }
        currentUser=null; sessionToken=null;
        sessionStorage.removeItem('sgd_token');
        document.getElementById('mainApp').classList.add('hidden');
        document.getElementById('loginScreen').style.display='flex';
        document.getElementById('chatFab').classList.add('hidden');
        document.getElementById('chatPanel').classList.add('hidden');
      }
    });
}

function actualizarSeriesTRD(){
  var depSelect = document.getElementById('deptoOrigen');
  var serieSelect = document.getElementById('serieTRD');
  if(!depSelect || !serieSelect) return;
  var depKey = depSelect.value || 'GE';
  
  var opciones = MAPA_TRD_DEPTOS[depKey] || [
    { val: '100-10.01', label: '100-10.01 — GERENCIA / Cartas y Comunicaciones (10 Años C.Co 60)' }
  ];
  
  var trdOptions = opciones.map(function(o){
    return '<option value="' + o.val + '">' + o.label + '</option>';
  }).join('');
    
  serieSelect.innerHTML = trdOptions;
  generarCodigoDoc();
}

function generarCodigoDoc(){
  var d=document.getElementById('deptoOrigen');
  var s=v('serieTRD');
  if(s && s !== ''){
    var parts=s.split('-');
    var depCode=parts[0];
    var serieSub=parts[1]||'';
    var subparts=serieSub.split('.');
    var serieCode=subparts[0];
    var subserieCode=subparts[1]||'';
    var depSigla = d ? d.value : 'GE';
    google.script.run.withSuccessHandler(function(c){ setVal('codigoDocumento',c); }).generarCodigoTRD(depSigla, depCode, serieCode, subserieCode);
  } else {
    setVal('codigoDocumento', 'Seleccione Serie TRD...');
  }
}

function iniciarPolling(){
  if(pollingInterval) clearInterval(pollingInterval);
  pollingInterval=setInterval(function(){
    if(!currentUser||document.visibilityState==='hidden') return;
    google.script.run.withSuccessHandler(function(version) {
      if (version === ultimaVersionSistema) {
        return; // No hay cambios, no recargar
      }
      ultimaVersionSistema = version;
      var sec=document.querySelector('.section.active');
      if(!sec) return;
      var id=sec.id;
      if(id==='correspondencia') cargarCorrespondencia();
      else if(id==='prestamos')  cargarPrestamos();
      else if(id==='dashboard')  cargarDashboard();
    }).obtenerCacheVersion();
  },30000);
}

// ============================================================
function renderCorr(){
  var datos=todasCorrespondencias.slice();
  var desde=v('filtroCorrDesde'),hasta=v('filtroCorrHasta'),origen=v('filtroCorrOrigen')||'todos',estado=v('filtroCorrEstado')||'todos';
  if(desde){ var d=new Date(desde); d.setHours(0,0,0,0); datos=datos.filter(function(i){ return new Date(i.fecha)>=d; }); }
  if(hasta){ var h=new Date(hasta); h.setHours(23,59,59,999); datos=datos.filter(function(i){ return new Date(i.fecha)<=h; }); }
  if(origen!=='todos') datos=datos.filter(function(i){ return i.origen===origen; });
  if(estado!=='todos') datos=datos.filter(function(i){ return i.estado===estado; });
  datos.sort(function(a,b){ return new Date(b.fecha)-new Date(a.fecha); });
  var total=datos.length,pages=Math.max(1,Math.ceil(total/POR_PAG_CORR));
  if(pagCorr>pages) pagCorr=pages;
  var start=(pagCorr-1)*POR_PAG_CORR, page=datos.slice(start,start+POR_PAG_CORR);
  document.getElementById('corrPagInfo').textContent='Pág. '+pagCorr+'/'+pages+' · '+total+' regs.';
  document.getElementById('corrPrev').disabled=pagCorr<=1;
  document.getElementById('corrNext').disabled=pagCorr>=pages;
  var el=document.getElementById('listaCorrespondencia');
  if(!page.length){ el.innerHTML='<div style="text-align:center;padding:36px;color:var(--text-muted)"><i class="fas fa-envelope-open" style="font-size:2.5rem;opacity:.15"></i><div style="margin-top:10px">Sin registros'+(todasCorrespondencias.length?' con los filtros aplicados':'')+'</div></div>'; return; }
  var bm={PENDIENTE:'badge-pending',ENVIADO:'badge-active',RECIBIDO:'badge-info',ENTREGADO:'badge-done',DEVUELTO:'badge-violet'};
  var im={PENDIENTE:'⏳',ENVIADO:'📤',RECIBIDO:'📥',ENTREGADO:'✅',DEVUELTO:'🔄'};

  // Opciones del selector de estado con descripción completa
  var opcionesEstado = [
    { val:'PENDIENTE', emoji:'⏳', label:'Pendiente',  desc:'En espera de proceso' },
    { val:'ENVIADO',   emoji:'📤', label:'Enviado',    desc:'Despachado al destino' },
    { val:'RECIBIDO',  emoji:'📥', label:'Recibido',   desc:'Confirmado en destino' },
    { val:'ENTREGADO', emoji:'✅', label:'Entregado',  desc:'Firmado y entregado' },
    { val:'DEVUELTO',  emoji:'🔄', label:'Devuelto',   desc:'Regresado al origen' }
  ];

  el.innerHTML='<div class="table-wrap"><table id="tablaCorr"><thead><tr><th>Fecha</th><th>Código</th><th>Tipo</th><th>Origen → Destino</th><th>Estado actual</th><th style="min-width:230px">Cambiar estado</th></tr></thead><tbody>'+
    page.map(function(i){
      var fs=i.fecha?new Date(i.fecha).toLocaleDateString('es-CO'):'—';
      var opHtml=opcionesEstado.map(function(op){
        var sel=(i.estado===op.val)?' selected':'';
        return '<option value="'+op.val+'"'+sel+'>'+op.emoji+' '+op.label+' — '+op.desc+'</option>';
      }).join('');
      var destBadge=i.destino&&i.destino!=='undefined'?'<span class="badge badge-violet" style="margin-left:4px">→ '+i.destino+'</span>':'';
      return '<tr>'+
        '<td style="white-space:nowrap">'+fs+'</td>'+
        '<td><code style="background:var(--bg-elevated);padding:2px 7px;border-radius:4px;font-size:.8rem">'+(i.codigo||'—')+'</code></td>'+
        '<td>'+(i.tipo||'—')+'</td>'+
        '<td><span class="badge badge-info">'+(i.origen||'—')+'</span>'+destBadge+'</td>'+
        '<td><span class="badge '+(bm[i.estado]||'badge-pending')+'">'+(im[i.estado]||'📄')+' '+(i.estado||'—')+'</span></td>'+
        '<td>'+
          '<div class="estado-select-wrap">'+
            '<select class="estado-select" data-codigo="'+i.codigo+'" data-estado="'+i.estado+'" onchange="actualizarEstadoCorr(\''+i.codigo+'\',this.value,this)">'+
              opHtml+
            '</select>'+
          '</div>'+
        '</td>'+
      '</tr>';
    }).join('')+'</tbody></table></div>';
}

// ============================================================
// actualizarEstadoCorr — VERSIÓN MEJORADA CON FEEDBACK VISUAL
// Guarda en Google Sheets y confirma con toast
// ============================================================
function actualizarEstadoCorr(cod, nuevoEstado, selectEl){
  // Bloquear el select mientras se guarda para evitar doble clic
  if(selectEl){
    selectEl.disabled=true;
    selectEl.style.opacity='0.45';
  }

  google.script.run
    .withSuccessHandler(function(r){
      // Reactivar el select
      if(selectEl){
        selectEl.disabled=false;
        selectEl.style.opacity='1';
      }

      if(r && r.success){
        // 1. Actualizar el array local en memoria
        for(var i=0;i<todasCorrespondencias.length;i++){
          if(todasCorrespondencias[i].codigo===cod){
            todasCorrespondencias[i].estado=nuevoEstado;
            break;
          }
        }
        // 2. Actualizar el atributo data-estado del propio select para cambiar el color del borde
        if(selectEl) selectEl.setAttribute('data-estado', nuevoEstado);

        // 3. Toast de confirmación — visible, rápido y no intrusivo
        Swal.fire({
          toast: true,
          position: 'top-end',
          icon: 'success',
          title: '✅ Estado actualizado en Google Sheets',
          html: '<small style="color:#94a3b8">Código <b>'+cod+'</b> → <b>'+nuevoEstado+'</b></small>',
          showConfirmButton: false,
          timer: 2800,
          timerProgressBar: true,
          background: '#ffffff',
          color: '#142a44',
          customClass:{ popup:'swal2-toast-coraza' }
        });

        // 4. Re-renderizar la fila para actualizar el badge de "Estado actual"
        renderCorr();

      } else {
        // Error del servidor — mostrar mensaje y revertir el select al valor anterior
        if(selectEl){
          // Revertir al valor anterior buscando en el array local
          for(var j=0;j<todasCorrespondencias.length;j++){
            if(todasCorrespondencias[j].codigo===cod){
              selectEl.value=todasCorrespondencias[j].estado;
              selectEl.setAttribute('data-estado',todasCorrespondencias[j].estado);
              break;
            }
          }
        }
        Swal.fire({
          toast: true,
          position: 'top-end',
          icon: 'error',
          title: '❌ No se pudo guardar',
          html: '<small>'+(r ? r.message||'Error en el servidor' : 'Sin respuesta')+'</small>',
          showConfirmButton: false,
          timer: 4000,
          background: '#ffffff',
          color: '#142a44'
        });
      }
    })
    .withFailureHandler(function(err){
      // Error de conexión
      if(selectEl){
        selectEl.disabled=false;
        selectEl.style.opacity='1';
        // Revertir al valor anterior
        for(var k=0;k<todasCorrespondencias.length;k++){
          if(todasCorrespondencias[k].codigo===cod){
            selectEl.value=todasCorrespondencias[k].estado;
            selectEl.setAttribute('data-estado',todasCorrespondencias[k].estado);
            break;
          }
        }
      }
      Swal.fire({
        toast: true,
        position: 'top-end',
        icon: 'error',
        title: '❌ Error de conexión',
        html: '<small>'+(err.message||'Intenta de nuevo')+'</small>',
        showConfirmButton: false,
        timer: 4000,
        background: '#ffffff',
        color: '#142a44'
      });
    })
    .actualizarEstadoCorrespondencia(cod, nuevoEstado);
}

function filtrarPrestamos(){
  var hoy=new Date(); hoy.setHours(0,0,0,0);
  var p=todosPrestamos.map(function(item){
    var fd=item.fechaDevolucion?new Date(item.fechaDevolucion):null; if(fd) fd.setHours(0,0,0,0);
    var dias=fd?Math.ceil((fd-hoy)/86400000):null;
    var ev=item.estado==='DEVUELTO'?'DEVUELTO':(fd&&fd<hoy?'VENCIDO':'ACTIVO');
    return Object.assign({},item,{diasRestantes:dias,estadoVisual:ev});
  });
  var es=v('filtroPrestEstado')||'todos', dp=v('filtroPrestDepto')||'todos';
  if(dp!=='todos') p=p.filter(function(i){ return i.departamento===dp; });
  if(es!=='todos') p=p.filter(function(i){ return i.estadoVisual===es; });
  p.sort(function(a,b){ return new Date(b.fechaPrestamo||0)-new Date(a.fechaPrestamo||0); });
  var el=document.getElementById('listaPrestamos');
  if(!p.length){ el.innerHTML='<div style="text-align:center;padding:36px;color:var(--text-muted)"><i class="fas fa-inbox" style="font-size:2.5rem;opacity:.15"></i><div style="margin-top:10px">Sin préstamos</div></div>'; return; }
  el.innerHTML='<div style="font-size:.78rem;color:var(--text-muted);margin-bottom:10px">'+p.length+' de '+todosPrestamos.length+' préstamos</div>'+
    '<div class="table-wrap"><table id="tablaPrest"><thead><tr><th>Estado</th><th>Solicitante</th><th>Depto</th><th>Documento</th><th>Devolución</th><th>Acción</th></tr></thead><tbody>'+
    p.slice(0,25).map(function(i){
      var cls='badge-active',txt='ACTIVO',icon='🟢';
      if(i.estadoVisual==='DEVUELTO'){cls='badge-done';txt='DEVUELTO';icon='✅';}
      else if(i.estadoVisual==='VENCIDO'){cls='badge-overdue';txt='VENCIDO';icon='🔴';}
      else if(i.diasRestantes!==null&&i.diasRestantes<=2){cls='badge-pending';txt='PRÓXIMO';icon='🟡';}

function ejecutarBusqueda(){
  var t=v('searchInput').trim(), f=v('filtroTipo')||'todos';
  var c=document.getElementById('resultadosBusqueda');
  
  // Limpiar resaltado previo en el mapa
  document.querySelectorAll('.slot').forEach(function(s){ s.classList.remove('highlight'); s.removeAttribute('title'); });
  
  if(!t||t.length<2){ c.innerHTML=''; document.getElementById('totalRes').textContent='0 resultados'; return; }
  c.innerHTML='<div style="text-align:center;padding:18px;color:var(--text-muted)"><i class="fas fa-spinner fa-spin"></i> Buscando...</div>';
  
  google.script.run.withSuccessHandler(function(r){
    if(!r || r.success === false){
      document.getElementById('totalRes').textContent='Error';
      c.innerHTML='<div style="text-align:center;padding:36px;color:var(--accent-red)"><i class="fas fa-exclamation-triangle"></i> Error: ' + (r ? r.message : 'Error de respuesta') + '</div>'; 
      return; 
    }
    if(!r.resultados || !r.resultados.length){ 
      document.getElementById('totalRes').textContent='0 resultados';
      c.innerHTML='<div style="text-align:center;padding:36px;color:var(--text-muted)">Sin resultados para "'+t+'"</div>'; 
      return; 
    }
    
    // Filtros cruzados avanzados
    var fi = v('searchFInicio');
    var ff = v('searchFFin');
    var depto = v('searchDepto').trim().toLowerCase();
    var estado = v('searchEstado').trim().toLowerCase();
    
    var resFiltrados = r.resultados;
    
    if (fi) {
      var d0 = new Date(fi + 'T00:00:00');
      resFiltrados = resFiltrados.filter(function(item) {
        if (!item.fecha) return true;
        return new Date(item.fecha) >= d0;
      });
    }
    if (ff) {
      var d1 = new Date(ff + 'T23:59:59');
      resFiltrados = resFiltrados.filter(function(item) {
        if (!item.fecha) return true;
        return new Date(item.fecha) <= d1;
      });
    }
    if (depto) {
      resFiltrados = resFiltrados.filter(function(item) {
        var desc = (item.descripcion || '').toLowerCase();
        var tit = (item.titulo || '').toLowerCase();
        return desc.indexOf(depto) !== -1 || tit.indexOf(depto) !== -1;
      });
    }
    if (estado) {
      resFiltrados = resFiltrados.filter(function(item) {
        var desc = (item.descripcion || '').toLowerCase();
        var tit = (item.titulo || '').toLowerCase();
        var tipo = (item.tipo || '').toLowerCase();
        return desc.indexOf(estado) !== -1 || tit.indexOf(estado) !== -1 || tipo.indexOf(estado) !== -1;
      });
    }

    window.lastSearchResults = resFiltrados;
    document.getElementById('totalRes').textContent = resFiltrados.length + ' resultados';
    if(resFiltrados.length === 0){
      c.innerHTML = '<div style="text-align:center;padding:36px;color:var(--text-muted)">Sin resultados que coincidan con los filtros avanzados</div>';
      return;
    }
    
    // Resaltar los slots del mapa que coincidan con las ubicaciones encontradas
    resFiltrados.forEach(function(d){
      if(d.ubicacion){
        var slotEl=document.getElementById('slot-'+d.ubicacion);
        if(slotEl){
          slotEl.classList.add('highlight');
          slotEl.title=d.tipo + ': ' + d.titulo + '\nCódigo: ' + (d.codigoUnico||'—');
        }
      }
    });

    var cols={Correspondencia:'var(--accent-primary)','Hoja de Vida':'var(--accent-amber)','Personal Activo':'var(--accent-amber)','Personal Inactivo':'var(--accent-amber)',Contrato:'var(--accent-violet)',Biblioteca:'var(--accent-cyan)',Préstamo:'var(--accent-red)',Minuta:'var(--accent-green)'};
    c.innerHTML='<div style="display:flex;flex-direction:column;gap:9px">'+
      resFiltrados.slice(0,10).map(function(d, idx){
        var badgeCod = d.codigoUnico ? '<span class="badge badge-done" style="font-family:monospace;background:rgba(74,116,167,.1);color:var(--accent-primary);font-size:.7rem">'+d.codigoUnico+'</span>' : '';
        return '<div style="display:flex;align-items:center;gap:14px;background:var(--bg-elevated);border:1px solid var(--border-medium);border-left:4px solid '+(cols[d.tipo]||'var(--accent-primary)')+';border-radius:var(--r-lg);padding:14px 18px">'+
          '<span style="font-size:1.6rem;width:46px;height:46px;background:'+(cols[d.tipo]||'var(--accent-primary)')+'20;border-radius:var(--r-md);display:flex;align-items:center;justify-content:center;flex-shrink:0">'+(d.icono||'📄')+'</span>'+
          '<div style="flex:1;min-width:0"><div style="display:flex;align-items:center;gap:8px;flex-wrap:wrap;margin-bottom:3px"><span style="font-weight:700;font-size:.92rem">'+(d.titulo||'')+'</span><span class="badge badge-info">'+d.tipo+'</span>'+badgeCod+'</div>'+
          '<div style="font-size:.8rem;color:var(--text-muted)">'+(d.descripcion||'')+'</div>'+
          (d.ubicacion?'<div style="font-size:.75rem;color:var(--accent-cyan);margin-top:3px"><i class="fas fa-map-marker-alt"></i> '+d.ubicacion+'</div>':'')+'</div>'+
          '<button class="btn btn-secondary btn-sm" onclick="verDetalleRegistro('+idx+')" style="padding: 6px 12px; font-size: .75rem; display:flex; align-items:center; gap:5px; flex-shrink:0;"><i class="fas fa-info-circle"></i> Detalles</button>'+
          '</div>';
      }).join('')+'</div>';
  }).withFailureHandler(function(){ c.innerHTML='<div class="alert alert-danger"><i class="fas fa-plug"></i>Error de conexión</div>'; }).busquedaUniversal(t,f);
}

function verDetalleRegistro(idx) {
  var item = window.lastSearchResults ? window.lastSearchResults[idx] : null;
  if (!item) return;
  
  var bodyHtml = '<div style="display:flex; flex-direction:column; gap:12px;">';
  
  bodyHtml += '<div style="display:flex; align-items:center; gap:12px; border-bottom:1px solid var(--border-medium); padding-bottom:12px; margin-bottom:8px;">' +
    '<span style="font-size:2rem; width:48px; height:48px; border-radius:var(--r-md); background:var(--accent-primary)15; display:flex; align-items:center; justify-content:center;">' + (item.icono || '📄') + '</span>' +
    '<div>' +
      '<div style="font-weight:700; font-size:1.15rem; color:var(--text-main); line-height:1.2;">' + (item.titulo || '') + '</div>' +
      '<span class="badge badge-info" style="margin-top:4px;">' + item.tipo + '</span>' +
    '</div>' +
  '</div>';
  
  bodyHtml += '<div style="display:grid; grid-template-columns:1fr; gap:9px; font-size:.85rem; max-height:55vh; overflow-y:auto; padding-right:4px;">';
  
  if (item.detalles) {
    for (var key in item.detalles) {
      if (item.detalles.hasOwnProperty(key)) {
        var val = item.detalles[key];
        if (val === null || val === undefined) val = '—';
        if (val instanceof Date || (typeof val === 'string' && val.indexOf('/Date(') !== -1)) {
          val = formatearFechaDetalle(val);
        }
        
        var displayKey = key.replace(/_/g, ' ');
        bodyHtml += '<div style="background:var(--bg-elevated); border:1px solid var(--border-medium); border-radius:var(--r-md); padding:8px 12px; display:flex; flex-direction:column; gap:3px;">' +
          '<span style="font-size:.65rem; color:var(--text-muted); font-weight:700; text-transform:uppercase; letter-spacing:0.5px;">' + displayKey + '</span>' +
          '<span style="font-weight:600; color:var(--text-main); word-break:break-all;">' + val + '</span>' +
        '</div>';
      }
    }
  } else {
    bodyHtml += '<div style="text-align:center; color:var(--text-muted); padding:16px;">Sin detalles disponibles.</div>';
  }
  
  bodyHtml += '</div></div>';
  
  document.getElementById('detalleBusquedaBody').innerHTML = bodyHtml;
  document.getElementById('modalDetalleBusqueda').classList.add('show');
}

function formatearFechaDetalle(val) {
  if (!val) return '—';
  try {
    var d = new Date(val);
    if (isNaN(d.getTime())) return val;
    return d.toLocaleDateString('es-CO', { year: 'numeric', month: '2-digit', day: '2-digit' }) + ' ' + d.toLocaleTimeString('es-CO', { hour: '2-digit', minute: '2-digit' });
  } catch(e) {
    return val;
  }
}

function buildChart(id, type, labels, datasets, opts){
  var ctx = document.getElementById(id);
  if(!ctx) return null;
  var defaults = {
    responsive: true,
    maintainAspectRatio: false,
    plugins: {
      legend: { labels: { color:'#5c728e', font:{ size:11 }, padding:14, boxWidth:12 } },
      tooltip: { backgroundColor:'#ffffff', titleColor:'#142a44', bodyColor:'#5c728e', borderColor:'#d0dfee', borderWidth:1, padding:10, cornerRadius:8 }
    },
    scales: type==='doughnut'||type==='pie' ? undefined : {
      x: { ticks:{ color:'#5c728e', font:{size:10} }, grid:{ color:'#e1ebf5' } },
      y: { ticks:{ color:'#5c728e', font:{size:10} }, grid:{ color:'#e1ebf5' }, beginAtZero:true }
    }
  };
  var finalOpts = Object.assign({}, defaults, opts||{});
  if(finalOpts.scales && opts && opts.scales) finalOpts.scales = Object.assign(defaults.scales||{}, opts.scales);
  var chart = new Chart(ctx, { type:type, data:{ labels:labels, datasets:datasets }, options:finalOpts });
  infCharts.push(chart);
  return chart;
}

function exportarTablaACSV(tablaId, nombreArchivo) {
  var table = document.getElementById(tablaId);
  if (!table) return;
  var rows = table.querySelectorAll('tr');
  var csvContent = '';
  for (var i = 0; i < rows.length; i++) {
    var cols = rows[i].querySelectorAll('th, td');
    var rowContent = [];
    for (var j = 0; j < cols.length; j++) {
      if (cols[j].querySelector('button') || cols[j].textContent.trim() === 'Acción' || cols[j].textContent.trim() === 'Acciones') {
        continue;
      }
      var text = cols[j].textContent.trim().replace(/"/g, '""');
      rowContent.push('"' + text + '"');
    }
    if (rowContent.length > 0) {
      csvContent += rowContent.join(';') + '\r\n';
    }
  }
  var blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' });
  var link = document.createElement('a');
  var url = URL.createObjectURL(blob);
  link.setAttribute('href', url);
  link.setAttribute('download', nombreArchivo + '.csv');
  link.style.visibility = 'hidden';
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

document.addEventListener('DOMContentLoaded',function(){
  var panel=document.getElementById('chatPanel');
  var handle=panel?panel.querySelector('.chat-top'):null;
  if(!handle) return;
  function start(cx,cy){ if(window.innerWidth<=600) return; isDragging=true; var r=panel.getBoundingClientRect(); dragOX=cx-r.left; dragOY=cy-r.top; panel.style.transition='none'; document.body.style.userSelect='none'; }
  function move(cx,cy){ if(!isDragging) return; var vw=window.innerWidth,vh=window.innerHeight; var nx=Math.max(0,Math.min(cx-dragOX,vw-panel.offsetWidth)); var ny=Math.max(0,Math.min(cy-dragOY,vh-panel.offsetHeight)); chatPos.x=nx; chatPos.y=ny; panel.style.left=nx+'px'; panel.style.top=ny+'px'; panel.style.right='auto'; panel.style.bottom='auto'; }
  function stop(){ isDragging=false; panel.style.transition=''; document.body.style.userSelect=''; }
  handle.addEventListener('mousedown',function(e){ start(e.clientX,e.clientY); });
  document.addEventListener('mousemove',function(e){ move(e.clientX,e.clientY); });
  document.addEventListener('mouseup',stop);
  handle.addEventListener('touchstart',function(e){ var t=e.touches[0]; start(t.clientX,t.clientY); },{passive:true});
  document.addEventListener('touchmove',function(e){ if(!isDragging) return; var t=e.touches[0]; move(t.clientX,t.clientY); },{passive:true});
  document.addEventListener('touchend',stop);
  window.addEventListener('resize',function(){ if(!panel.classList.contains('hidden')){ chatPos.x=null; chatPos.y=null; initChatPos(); } });
});
